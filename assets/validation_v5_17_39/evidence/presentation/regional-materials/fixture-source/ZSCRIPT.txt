    override void NetworkProcess(ConsoleEvent e)
    {
        if (level.MapName != "CA709G") return;
        if (e.Name == "ca709_prepare")
        {
            if (Subjects.Size() != 0) return;
            CaelumPlayer(players[0].mo).CharacterCreationComplete = true;
            CaelumPlayer(players[0].mo).CreationWizardOpen = false;
            players[0].mo.bNOTARGET=true;players[0].mo.bInvulnerable=true;
            ObserverNoiseSerial=CaelumPlayer(players[0].mo).MovementNoiseEventSerial;
            {let item=CaelumBestiaryCombatActor(Actor.Spawn('CaelumBestiary006',(-768,0,0),NO_REPLACE));if(item!=null){item.CaelumDiagnosticPassiveAI=true;item.bFriendly=true;item.Angle=270;}Subjects.Push(item);}

{let item=CaelumBestiaryCombatActor(Actor.Spawn('CaelumBestiary051',(-128,256,0),NO_REPLACE));if(item!=null){item.CaelumDiagnosticPassiveAI=true;item.bFriendly=true;item.Angle=270;}Subjects.Push(item);}
            View = CA709Camera(Actor.Spawn('CA709Camera', (0,0,0), NO_REPLACE));
            ConfigureView(0);
        }
        if (e.Name == "ca709_view") ConfigureView(e.Args[0]);
        if (e.Name == "ca709_verify") Verify();
        if (e.Name == "ca709_motion_start")
        {
            MotionSnapshot.Clear();
            for (int index=0; index<Subjects.Size(); index++) MotionSnapshot.Push(Subjects[index].CurState);
        }
        if (e.Name == "ca709_motion_verify")
        {
            Check(MotionSnapshot.Size() == Subjects.Size(), "animation snapshot registry");
            for (int index=0; index<Subjects.Size(); index++)
                Check(Subjects[index].CurState != MotionSnapshot[index] && IsIdleState(Subjects[index],MotionSnapshot[index]) && IsIdleState(Subjects[index],Subjects[index].CurState), String.Format("subject %d native idle advancement", index));
            Console.Printf("CA709 COMPLETE failures=%d", Failures);
        }
        if (e.Name == "ca709_walk_start")
        {
            ConfigureView(4);
            WalkStart = players[0].mo.Pos;
        }
        if (e.Name == "ca709_walk_verify")
        {
            let user=players[0].mo;
            Console.Printf("CA709 WALK start=(%.3f,%.3f,%.3f) end=(%.3f,%.3f,%.3f) angle=%.3f",WalkStart.X,WalkStart.Y,WalkStart.Z,user.Pos.X,user.Pos.Y,user.Pos.Z,user.Angle);
            Check((user.Pos-WalkStart).Length() > user.Radius, "native player movement in gallery");
            Check(abs(user.Pos.Z-user.floorz)<0.01, "player remains on the walkable floor");
            Console.Printf("CA709 COMPLETE failures=%d", Failures);
        }
        if (e.Name == "ca709_cross_start")
        {
            // Recorrido diagnóstico de las 3 uniones entre parcelas.
            players[0].mo.SetOrigin((-1280,-256,0), false);
            players[0].mo.Angle = 90;
            players[0].mo.Vel = (0,0,0);
            Console.Printf("CA709 CROSS START pos=(%.3f,%.3f,%.3f) angle=%.3f",players[0].mo.Pos.X,players[0].mo.Pos.Y,players[0].mo.Pos.Z,players[0].mo.Angle);
        }
        if (e.Name == "ca709_cross_verify")
        {
            let user=players[0].mo;
            Console.Printf("CA709 CROSS END pos=(%.3f,%.3f,%.3f) angle=%.3f",user.Pos.X,user.Pos.Y,user.Pos.Z,user.Angle);
            Check(user.Pos.X > 0, "native movement crosses all 3 regional boundaries");
            Check(abs(user.Pos.Z-user.floorz)<0.01, "regional crossing stays on the floor");
            Console.Printf("CA709 COMPLETE failures=%d", Failures);
        }
    }
