version "4.14.2"
// #647: objetos reales en una sala aislada; no altera reglas de inventario.
class CA647CoinCamera : Actor
{
    Default { CameraHeight 0; Height 0; Radius 0; +NOINTERACTION +NOGRAVITY }
    States { Spawn: TNT1 A -1; Stop; }
}

class CA647CoinEvents : EventHandler
{
    Array<CaelumCurrencyItem> Drops;
    Array<CaelumCurrencyItem> OwnedStacks;
    CA647CoinCamera Camera;
    int Failures;

    void Check(bool okay, String label)
    {
        if (!okay) Failures++;
        Console.Printf("CA647 %s %s", okay ? "PASS" : "FAIL", label);
    }

    void Verify()
    {
        Check(Drops.Size() == 15 && OwnedStacks.Size() == 15, "all denominations retained");
        for (int index = 0; index < Drops.Size(); index++)
        {
            let item = Drops[index];
            let owned = OwnedStacks[index];
            String label = String.Format("coin %d", index);
            Check(item != null && owned != null, label .. " saved references");
            if (item == null || owned == null) continue;
            Check(item.Owner == null && owned.Owner == players[0].mo,
                label .. " world/inventory ownership");
            Check(item.Amount == 7 && owned.Amount == 30, label .. " split quantities");
            Check(item.GetClass() == owned.GetClass()
                && item.GetCurrencyType() == owned.GetCurrencyType()
                && item.GetFaceValue() == owned.GetFaceValue()
                && item.GetNominalDenomination() == owned.GetNominalDenomination(),
                label .. " denomination identity");
            Check(item.GetUnitWeight() == CaelumConstants.CURRENCY_UNIT_WEIGHT,
                label .. " unit weight");
            let defaults = GetDefaultByType(item.GetClass());
            Check(item.Radius == defaults.Radius && item.Height == defaults.Height
                && item.Scale == defaults.Scale, label .. " native collision and scale");
            Check(item.Icon == owned.Icon && item.sprite == owned.sprite,
                label .. " original icon and state");
            Check(abs(item.Pos.Z - item.floorz) < 0.01, label .. " rests on floor");
        }
        Console.Printf("CA647 COMPLETE failures=%d", Failures);
    }

    override void NetworkProcess(ConsoleEvent e)
    {
        if (e.Name == "ca647_prepare")
        {
            static const class<CaelumCurrencyItem> kinds[] = { 'CaelumCopperCoin','CaelumCopperCoin5','CaelumCopperCoin20','CaelumCopperCoin50','CaelumCopperCoin100','CaelumSilverCoin','CaelumSilverCoin5','CaelumSilverCoin20','CaelumSilverCoin50','CaelumSilverCoin100','CaelumGoldCoin','CaelumGoldCoin5','CaelumGoldCoin20','CaelumGoldCoin50','CaelumGoldCoin100' };
            Actor user = players[0].mo;
            for (int index = 0; index < kinds.Size(); index++)
            {
                let incoming = CaelumCurrencyItem(Actor.Spawn(kinds[index], (0,0,0), NO_REPLACE));
                incoming.Amount = 37;
                Check(incoming.CallTryPickup(user), String.Format("coin %d native pickup", index));
                let owned = CaelumCurrencyItem(user.FindInventory(kinds[index]));
                Check(owned != null && owned.Amount == 37, String.Format("coin %d owned quantity", index));
                if (owned == null) continue;
                let dropped = CaelumCurrencyItem(owned.CreateTossable(7));
                Check(dropped != null, String.Format("coin %d native toss", index));
                if (dropped == null) continue;
                dropped.SetOrigin(((index % 5 - 2) * 20, (index / 5 - 1) * 20, 0), false);
                Drops.Push(dropped);
                OwnedStacks.Push(owned);
            }
            Camera = CA647CoinCamera(Actor.Spawn('CA647CoinCamera', (0,-100,110), NO_REPLACE));
            Camera.Angle = 90;
            Camera.Pitch = 47;
            players[0].camera = Camera;
            players[0].mo.bInvisible = true;
        }
        if (e.Name == "ca647_view")
        {
            if (e.Args[0] == 0) { Camera.SetOrigin((0,-100,110), false); Camera.Angle=90; Camera.Pitch=47; }
            if (e.Args[0] == 1) { Camera.SetOrigin((0,-85,10), false); Camera.Angle=90; Camera.Pitch=6; }
            if (e.Args[0] == 2) { Camera.SetOrigin((75,0,80), false); Camera.Angle=180; Camera.Pitch=47; }
            if (e.Args[0] == 3) { Camera.SetOrigin((0,-40,22), false); Camera.Angle=90; Camera.Pitch=43; }
            Camera.ClearInterpolation();
        }
        if (e.Name == "ca647_verify") Verify();
        if (e.Name == "ca647_pickup")
        {
            Actor user = players[0].mo;
            for (int index = 0; index < Drops.Size(); index++)
            {
                Check(Drops[index].CallTryPickup(user), String.Format("coin %d repickup", index));
                Check(OwnedStacks[index].Amount == 37 && OwnedStacks[index].Owner == user,
                    String.Format("coin %d combined native stack", index));
            }
            Console.Printf("CA647 COMPLETE failures=%d", Failures);
        }
    }
}
