def uv_rect(tile,u,v):
    x0,y0,x1,y1=DESIGN['reference_crops'][tile]
    return ((x0+u*(x1-x0))/image.size[0],1-(y0+(1-v)*(y1-y0))/image.size[1])

def mesh(name,verts,faces,bone,tile='metal',uvs=None,head=False):
    data=bpy.data.meshes.new(name);data.from_pydata(verts,[],faces);data.update()
    obj=bpy.data.objects.new(name,data);scene.collection.objects.link(obj)
    data.materials.append(material)
    uv=data.uv_layers.new(name='Original-sheet UV')
    coords=[Vector(p) for p in verts]
    mn=Vector(tuple(min(p[i] for p in coords) for i in range(3)))
    mx=Vector(tuple(max(p[i] for p in coords) for i in range(3)))
    for poly in data.polygons:
        for li in poly.loop_indices:
            vi=data.loops[li].vertex_index;p=coords[vi]
            if uvs is not None:uv.data[li].uv=uv_rect(tile,*uvs[vi]);continue
            which=tile
            if head:
                which='face_front' if poly.normal.x>0.2 else 'hair_back' if poly.normal.x<-.2 else 'face_profile'
            axis=(0 if which=='face_profile' else 1) if head else (1 if abs(poly.normal.x)>=abs(poly.normal.y) else 0)
            u=(p[axis]-mn[axis])/max(.001,mx[axis]-mn[axis])
            v=(p.z-mn.z)/max(.001,mx.z-mn.z)
            uv.data[li].uv=uv_rect(which,u,v)
        poly.use_smooth=True
    group=obj.vertex_groups.new(name=bone);group.add(list(range(len(verts))),1,'REPLACE')
    part=obj.vertex_groups.new(name='Part.'+name);part.add(list(range(len(verts))),1,'REPLACE')
    mod=obj.modifiers.new('Reusable rig','ARMATURE');mod.object=arm
    obj.parent=arm;parts.append(obj)
    return obj

def ellipsoid(name,center,radii,bone,tile='metal',segments=12,rings=6,head=False):
    verts=[];faces=[]
    for r in range(rings+1):
        phi=math.pi*(r+.01)/(rings+.02)
        for j in range(segments):
            a=j*2*math.pi/segments
            verts.append((center[0]+radii[0]*math.sin(phi)*math.cos(a),
                          center[1]+radii[1]*math.sin(phi)*math.sin(a),
                          center[2]+radii[2]*math.cos(phi)))
    if head:
        verts=[(x+4*max(0,1-abs(y)/3)*max(0,1-abs(z-157.5)/5) if x>3 else x,y,z) for x,y,z in verts]
    for r in range(rings):
        for j in range(segments):
            a=r*segments+j;b=r*segments+(j+1)%segments
            faces.append((a,b,b+segments,a+segments))
    faces+=[tuple(reversed(range(segments))),tuple(rings*segments+j for j in range(segments))]
    return mesh(name,verts,[tuple(reversed(f)) for f in faces],bone,tile,head=head)

def beam(name,a,b,rx,ry,bone,tile):
    a=Vector(a);b=Vector(b);direction=b-a
    q=Vector((0,0,1)).rotation_difference(direction.normalized())
    verts=[];faces=[]
    for t,factor in [(0,.75),(.12,1),(.85,.9),(1,.7)]:
        for j in range(10):
            angle=j*math.tau/10
            verts.append(a+direction*t+q@Vector((rx*math.cos(angle)*factor,ry*math.sin(angle)*factor,0)))
    for r in range(3):
        for j in range(10):faces.append((r*10+j,r*10+(j+1)%10,(r+1)*10+(j+1)%10,(r+1)*10+j))
    faces.extend([tuple(reversed(range(10))),tuple(range(30,40))])
    return mesh(name,verts,faces,bone,tile)

def rotate(name,axis,angle):
    b=arm.pose.bones[name]
    local=b.bone.matrix_local.to_quaternion().inverted()@Vector(axis)
    b.rotation_quaternion=Quaternion(local,math.radians(angle))

def translate(name,position):
    b=arm.pose.bones[name]
    b.location=b.bone.matrix_local.to_3x3().inverted()@Vector(position)

def pose(action,t):
    for b in arm.pose.bones:b.rotation_mode='QUATERNION';b.rotation_quaternion=(1,0,0,0);b.location=(0,0,0)
    grounded=action not in ('LieDown','Sleeping','GetUp','Death','Raise')
    for side in ['L','R']:
        arm.pose.bones['shin.'+side].constraints['Foot plant IK'].influence=float(grounded)
        arm.pose.bones['foot.'+side].constraints['Foot orientation'].influence=float(grounded)
    wave=math.sin(t*math.tau)
    rotate('chest',(0,1,0),.7*wave)
    if action in ('Walk','Run'):
        amplitude=28 if action=='Walk' else 48
        for side,sign in [('L',1),('R',-1)]:
            phase=(t+(0 if side=='L' else .5))%1
            stance=.6 if action=='Walk' else .35
            excursion=52 if action=='Walk' else 90
            lift=12 if action=='Walk' else 25
            if phase<stance:x=excursion*(.5-phase/stance);z=0
            else:
                swing=(phase-stance)/(1-stance);x=excursion*(-.5+swing);z=lift*math.sin(math.pi*swing)
            translate('foot_ik.'+side,(x,0,z))
            rotate('upper_arm.'+side,(0,1,0),-amplitude*.55*wave*sign)
            rotate('forearm.'+side,(0,1,0),-15)
        translate('root',(0,0,-6+abs(wave)*2 if action=='Walk' else -14+abs(wave)*8))
        rotate('cape.lower',(0,1,0),4*wave)
    elif action in ('Melee','Missile'):
        swing=math.sin(math.pi*t)
        rotate('chest',(0,0,1),-25*swing)
        rotate('upper_arm.R',(0,1,0),-110*swing)
        rotate('forearm.R',(0,1,0),-35*swing)
        rotate('upper_arm.L',(0,1,0),-30*swing)
        rotate('sword',(1,0,0),25*swing)
    elif action=='Pain':rotate('spine',(0,1,0),-12*math.sin(math.pi*t))
    elif action in ('SitDown','Seated','StandUp'):
        f=t if action=='SitDown' else 1-t if action=='StandUp' else 1
        translate('root',(0,0,-(79-DESIGN['authoring']['seat_support_units'])*f))
        translate('cape.lower',(-40*f,0,30*f))
        for side in ['L','R']:
            translate('foot_ik.'+side,(40*f,0,0))
            rotate('upper_arm.'+side,(0,1,0),-25*f);rotate('forearm.'+side,(0,1,0),-45*f)
    elif action in ('LieDown','Sleeping','GetUp','Death','Raise'):
        f=t if action in ('LieDown','Death') else 1-t if action in ('GetUp','Raise') else 1
        death=action in ('Death','Raise')
        rotate('root',(0,1,0),(90 if death else -90)*f)
        translate('root',((-85 if death else 85)*f,0,20*f))
        if death:
            rotate('spine',(0,1,0),30*math.sin(math.pi*f))
            for side in ['L','R']:
                rotate('thigh.'+side,(0,1,0),-55*math.sin(math.pi*f));rotate('shin.'+side,(0,1,0),90*math.sin(math.pi*f))
        for side,sign in [('L',1),('R',-1)]:rotate('upper_arm.'+side,(1,0,0),sign*15*f)
    elif action=='Crouch':
        translate('root',(0,0,-42))
        rotate('spine',(0,1,0),22)
